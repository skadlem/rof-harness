```rust
// src/tui/app.rs
use crate::obs::TraceEvent;
use super::render::{render_line, Counters};

#[derive(Debug, Default)]
pub struct App {
    pub transcript: Vec<String>,
    pub counters: Counters,
    pub input: String,
    pub scroll: usize,
}

impl App {
    pub fn new() -> Self {
        Self::default()
    }

    /// Sole writer of transcript + counters. Plan C calls this live;
    /// replay mode calls this per file line.
    pub fn on_event(&mut self, ev: &TraceEvent) {
        self.transcript.push(render_line(ev));
        self.counters = Counters::fold(&[ev.clone(), ..[]]); // placeholder-free: see note
    }

    pub fn status_line(&self) -> String {
        let c = &self.counters;
        format!("calls={} in={} out={} pass={} fail={}", c.model_calls, c.in_tokens, c.out_tokens, c.pass, c.fail)
    }
}
```

NOTE — do NOT implement `fold` incrementally as sketched above (it would reset counters per event). Correct minimal form, write this instead:

```rust
pub fn on_event(&mut self, ev: &TraceEvent) {
    self.transcript.push(render_line(ev));
    match ev {
        TraceEvent::ModelCall { input_tokens, output_tokens, cost_usd, .. } => {
            self.counters.model_calls += 1;
            self.counters.in_tokens += input_tokens;
            self.counters.out_tokens += output_tokens;
            self.counters.cost_usd += cost_usd.unwrap_or(0.0);
        }
        TraceEvent::ReviewVerdict { pass, .. } => {
            if *pass { self.counters.pass += 1; } else { self.counters.fail += 1; }
        }
        _ => {}
    }
}
```

(`Counters` fields are `pub` per Plan A Task 1 — verify before writing; if not, add a `Counters::add_event(&mut self, ev)` method in render.rs instead and call it here.)

```rust
// src/tui/ui.rs
use ratatui::{
    layout::{Constraint, Direction, Layout},
    widgets::{Block, Borders, Paragraph},
    Frame,
};
use super::app::App;

/// Transcript (top) · status (thin middle) · composer (bottom, 3 lines).
/// Logic-free: every string comes from `App`.
pub fn draw(f: &mut Frame, app: &App) {
    let rows = Layout::default()
        .direction(Direction::Vertical)
        .constraints([
            Constraint::Min(1),
            Constraint::Length(1),
            Constraint::Length(3),
        ])
        .split(f.area());
    let tail: Vec<String> = app.transcript.iter().rev().take(rows[0].height as usize).collect();
    let shown: Vec<String> = tail.into_iter().rev().collect();
    f.render_widget(Paragraph::new(shown.join("\n")), rows[0]);
    f.render_widget(Paragraph::new(app.status_line()), rows[1]);
    f.render_widget(
        Paragraph::new(app.input.as_str()).block(Block::default().borders(Borders::ALL)),
        rows[2],
    );
}
```
